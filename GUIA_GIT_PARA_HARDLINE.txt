# GUÍA DE GIT PARA HARDLINE
# ================================

## INSTALACIÓN DE GIT
1. Descargar Git desde: https://git-scm.com/download/win
2. Instalar con opciones recomendadas:
   - Add Git Bash to PATH
   - Use Git from Windows Command Prompt

## CONFIGURACIÓN INICIAL
powershell
# Configurar usuario
git config --global user.name "Tu Nombre"
git config --global user.email "tu@email.com"

# Ver configuración
git config --list

## CLONAR REPOSITORIO DE HARDLINE
powershell
# Desde el directorio donde quieres clonar (ej: C:\Users\Nombre)
cd C:\Users\Nombre
git clone https://github.com/adrianlunamx/Hardline.git

# Entrar al directorio clonado
cd Hardline

## HACER PULL (Actualizar desde GitHub)
powershell
# Navegar al directorio del proyecto
cd C:\Users\Adrian L\AppData\Local\Hardline

# Si ya es un repositorio Git configurado:
git pull origin main

# Si necesitas especificar rama:
git pull origin master

## CONFIGURAR REMOTO (Si no está configurado)
powershell
# Primero inicializar git en tu proyecto local
git init

# Agregar todos los archivos
git add .

# Hacer primer commit
git commit -m "Initial commit"

# Agregar remoto de GitHub
git remote add origin https://github.com/adrianlunamx/Hardline.git

# Verificar remotos
git remote -v

## VER ESTADO DEL REPOSITORIO
powershell
# Ver archivos modificados
git status

# Ver historial de commits
git log --oneline

# Ver ramas
git branch -a

## GESTIONAR CAMBIOS LOCALES ANTES DE PULL
powershell
# 1. Guardar cambios locales primero
git add .
git commit -m "Guardar cambios locales"

# 2. Hacer pull
git pull origin main

# 3. Si hay conflictos, resolverlos y luego:
git add archivo.conflicto
git commit -m "Resuelto conflicto"

## COMANDOS ÚTILES
powershell
# Deshacer cambios no guardados en archivo
git checkout -- archivo.txt

# Eliminar archivo del tracking
git rm archivo.txt

# Crear nueva rama
git branch nueva-rama

# Cambiar de rama
git checkout nueva-rama

## ERRORES COMUNES Y SOLUCIONES

### "fatal: not a git repository"
# No estás en un directorio con git inicializado
git init

### "remote origin already exists"
# Ya tienes un remoto llamado origin
git remote -v                 # Ver remotos
git remote rename origin old  # Renombrar si necesitas
git remote add origin https://github.com/usuario/repo.git

### "Please tell me who you are"
# No has configurado usuario/email
git config --global user.name "Tu Nombre"
git config --global user.email "tu@email.com"

## FLUJO DE TRABAJO RECOMENDADO
1. git status                    # Ver estado
2. git add .                     # Preparar cambios
3. git commit -m "Mensaje"      # Confirmar cambios
4. git pull origin main          # Actualizar del servidor
5. Resolver conflictos si los hay
6. git push origin main          # Subir cambios
